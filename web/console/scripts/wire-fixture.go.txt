type wireConsole struct{ fakeConsole }

func (*wireConsole) SubmissionCapabilities() (bool, bool) { return true, true }
func (*wireConsole) Submit(_ context.Context, req consoleapi.Submission) (consoleapi.Exchange, error) {
	return consoleapi.Exchange{
		ID: "ack", Conversation: req.Conversation, Input: req.Input, Key: "client:" + req.CommandID,
		Quotes: req.Quotes, State: consoleapi.ExchangeQueued, History: "private agent history",
		EnqueuedAt: time.Date(2026, 9, 19, 0, 0, 0, 0, time.UTC),
	}, nil
}

type wireResponse struct {
	Status int    `json:"status"`
	Body   string `json:"body"`
	Cache  string `json:"cache"`
}

func TestConsoleWireFixture(t *testing.T) {
	destination := os.Getenv("CONSOLE_WIRE_FIXTURE")
	if destination == "" {
		t.Fatal("CONSOLE_WIRE_FIXTURE is required")
	}
	states := []consoleapi.ExchangeState{
		consoleapi.ExchangeQueued, consoleapi.ExchangeRunning, consoleapi.ExchangeRecovering,
		consoleapi.ExchangeAwaitingUser, consoleapi.ExchangeDone, consoleapi.ExchangeFailed, consoleapi.ExchangeCancelled,
	}
	at := time.Date(2026, 9, 19, 0, 0, 0, 0, time.UTC)
	conversation := "console:wire"
	owner := &wireConsole{}
	for _, state := range states {
		owner.exchanges = append(owner.exchanges, consoleapi.Exchange{
			ID: string(state), Conversation: conversation, Input: "wire input",
			State: state, Key: "client:" + string(state), EnqueuedAt: at, StartedAt: at,
		})
	}
	server := &Server{console: owner}
	capture := func(method, path, body string, handler http.HandlerFunc) wireResponse {
		w := httptest.NewRecorder()
		handler(w, httptest.NewRequest(method, path, strings.NewReader(body)))
		return wireResponse{w.Code, w.Body.String(), w.Header().Get("Cache-Control")}
	}
	queue := capture("GET", "/console/queue?conversation="+conversation, "", server.consoleQueue)
	ack := capture("POST", "/console/queue", `{"conversation":"console:wire","input":"wire input","command_id":"wire-command","quotes":[{"conversation":"console:source","reply_id":"r1"}]}`, server.consoleEnqueue)
	// A console honouring neither optional submission field, and the nil -> []
	// normalization, use the same route.
	basic := &Server{console: &fakeConsole{}}
	empty := capture("GET", "/console/queue", "", basic.consoleQueue)
	unwired := &Server{}
	plain := capture("GET", "/console/queue", "", unwired.consoleQueue)
	errorResponses := map[string]wireResponse{"unwired": plain}
	for name, err := range map[string]error{
		"invalid": errors.New("invalid input"), "missing": consoleapi.ErrExchangeNotFound,
		"not_queued": consoleapi.ErrExchangeNotQueued, "conflict": consoleapi.ErrCommandConflict,
		"closing": consoleapi.ErrConsoleClosing, "rewound": consoleapi.ErrRewindTargetGone,
	} {
		errorResponses[name] = capture("GET", "/", "", func(w http.ResponseWriter, _ *http.Request) { queueResponse(w, nil, err) })
	}
	events := []readmodel.Event{
		{At: at, Kind: "console.sent", Conversation: conversation, ExchangeID: "running", ReplyID: "sent", Text: "wire input"},
		{At: at.Add(time.Second), Kind: "console.progress", Conversation: conversation, ExchangeID: "running", Progress: &consoleapi.Progress{Answer: "partial", Phase: "working"}},
		{At: at.Add(2 * time.Second), Kind: "delegate.progress", Conversation: conversation, TaskID: "child", StepID: "#child",
			Step: &consoleapi.StepProcess{ID: "#child", StepInfo: consoleapi.StepInfo{State: "running", Answer: "child partial"}}},
		{At: at.Add(3 * time.Second), Kind: "console.step", Conversation: conversation, TaskID: "child",
			Step: &consoleapi.StepProcess{ID: "#child", StepInfo: consoleapi.StepInfo{State: "done", Answer: "child done"}}},
		{At: at.Add(4 * time.Second), Kind: "console.reply", Conversation: conversation, ExchangeID: "running", ReplyID: "reply", Text: "wire done", Format: "markdown"},
		{At: at.Add(5 * time.Second), Kind: "console.queue", Conversation: conversation},
		{At: at.Add(6 * time.Second), Kind: "console.notice", Conversation: conversation, ReplyID: "notice", Text: "temporary"},
		{At: at.Add(7 * time.Second), Kind: "console.recalled", Conversation: conversation, ReplyID: "notice"},
		{At: at.Add(8 * time.Second), Kind: "run.finished", RunID: "run", Seq: 7, TaskID: "child", Rev: 2, State: "done"},
	}
	stream := httptest.NewRecorder()
	for i, event := range events {
		writeEvent(stream, "wire."+strconv.Itoa(i+1), event)
	}
	out := map[string]any{
		"queue": queue, "empty": empty, "ack": ack, "errors": errorResponses,
		"events": events, "sse": stream.Body.String(),
	}
	// The owner keeps the old receipt when a keyed recovery-stop is retried.
	// Exercise its actual wire identity, not an invented frontend attempt ID.
	owner.exchanges = []consoleapi.Exchange{{
		ID: "stop", Conversation: conversation, Input: "/cancel", Key: "client:stop",
		State: consoleapi.ExchangeFailed, EnqueuedAt: at, StartedAt: at, ReplyID: "failed-stop",
	}}
	retry := map[string]wireResponse{}
	for _, state := range []consoleapi.ExchangeState{consoleapi.ExchangeFailed, consoleapi.ExchangeRunning, consoleapi.ExchangeDone} {
		owner.exchanges[0].State = state
		if state == consoleapi.ExchangeDone {
			owner.exchanges[0].ReplyID = "retry-done"
		}
		retry[string(state)] = capture("GET", "/console/queue?conversation="+conversation, "", server.consoleQueue)
	}
	out["retry"] = retry
	out["retry_replies"] = []consoleapi.Reply{{ID: "failed-stop", ExchangeID: "stop", Conversation: conversation, Kind: "reply", At: at, Text: "old stop failure"}}
	nano := httptest.NewRecorder()
	for _, event := range []readmodel.Event{
		{At: at, Kind: "console.sent", Conversation: conversation, ExchangeID: "nano"},
		{At: at.Add(123900001 * time.Nanosecond), Kind: "console.progress", Conversation: conversation, ExchangeID: "nano", Progress: &consoleapi.Progress{Answer: "newer"}},
		{At: at.Add(123900000 * time.Nanosecond).In(time.FixedZone("offset", 8*60*60)), Kind: "console.progress", Conversation: conversation, ExchangeID: "nano", Progress: &consoleapi.Progress{Answer: "older"}},
	} {
		writeEvent(nano, "nano.1", event)
	}
	out["nano_sse"] = nano.Body.String()
 out["work"] = workWireFixture(t, at)
	raw, err := json.Marshal(out)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(destination, raw, 0o600); err != nil {
		t.Fatal(err)
	}
}
